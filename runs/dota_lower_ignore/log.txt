#!/bin/bash
#SBATCH --job-name=lower-all-ignore
#SBATCH --time=24:00:00
#SBATCH --nodes=1
#SBATCH --ntasks-per-node=1
#SBATCH --cpus-per-task=16
#SBATCH --mem=64G
#SBATCH --partition=gpu_a100 
#SBATCH --gpus-per-node=1
#SBATCH --gres=gpu:1
#SBATCH --output=logs/lower-all-ignore-%j.out
#SBATCH --error=logs/lower-all-ignore-%j.err

module purge
module load 2023
module load CUDA/12.8.0
module load Python/3.11.3-GCCcore-12.3.0

source $HOME/ednet_env/bin/activate
cd $HOME/scriptie

python train.py \
  --cfg $HOME/scriptie/cfg/models/ednet/ednet-s.yaml \
  --data $HOME/scriptie/yaml/DOTA-filtered.yaml \
  --epochs 50 \
  --imgsz 640 \
  --batch-size 4 \
  --project $HOME/scriptie/runs/dota_lower_ignore \
  --name stage1-lower-ignore \
  --workers 8 \
  --weights pretrained/small.pt \
  --override "stage.active_classes=[0,1]" \
  --override "stage.prev_classes=[]"

python train.py \
  --cfg $HOME/scriptie/cfg/models/ednet/ednet-s.yaml \
  --data $HOME/scriptie/yaml/DOTA-filtered.yaml \
  --epochs 50 \
  --imgsz 640 \
  --batch-size 4 \
  --project $HOME/scriptie/runs/dota_lower_ignore \
  --name stage2-lower-ignore \
  --workers 8 \
  --weights /home/ddwilde/scriptie/runs/dota_lower_ignore/stage1-lower-ignore/weights/best.pt \
  --override "stage.active_classes=[2,3]" \
  --override "stage.prev_classes=[0,1]"

python train.py \
  --cfg $HOME/scriptie/cfg/models/ednet/ednet-s.yaml \
  --data $HOME/scriptie/yaml/DOTA-filtered.yaml \
  --epochs 50 \
  --imgsz 640 \
  --batch-size 4 \
  --project $HOME/scriptie/runs/dota_lower_ignore \
  --name stage3-lower-ignore \
  --workers 8 \
  --weights /home/ddwilde/scriptie/runs/dota_lower_ignore/stage2-lower-ignore/weights/best.pt \
  --override "stage.active_classes=[4]" \
  --override "stage.prev_classes=[0,1,2,3]"
